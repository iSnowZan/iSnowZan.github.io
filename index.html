package main

import (
  "context"
  "net/http"
  "log/slog"
)

func main() {
  db  := mustConnectPostgres(ctx)
  mq  := mustConnectRabbitMQ(ctx)
  rdb := mustConnectRedis(ctx)

  srv := NewServer(db, mq, rdb)
  slog.Info("listening", "port", 8080)
  http.ListenAndServe(":8080", srv)
}
